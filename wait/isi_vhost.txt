# Virtual Host untuk WebRST (Redirect HTTP ke HTTPS)
<VirtualHost *:80>
ServerName rstdrasmirsalatiga.co.id
Redirect permanent / https://rstdrasmirsalatiga.co.id/
</VirtualHost>
 
# Virtual Host untuk WebRST (HTTPS)
<VirtualHost *:443>
DocumentRoot "C:/xampp/htdocs/webRST/public"
ServerName rstdrasmirsalatiga.co.id
     
<Directory "C:/xampp/htdocs/webRST/public">
AllowOverride All
Require all granted
</Directory>
 
 SSLEngine on
 SSLCertificateFile "C:/xampp/htdocs/webRST/.well-known/fullchain.pem" 
 SSLCertificateKeyFile "C:/xampp/htdocs/webRST/.well-known/privkey.pem"
 </VirtualHost>
